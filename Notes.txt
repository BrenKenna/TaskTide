import jakarta.nosql.Column;
import jakarta.nosql.Entity;
import jakarta.nosql.Id;
import jakarta.json.bind.annotation.JsonbProperty;
import jakarta.json.bind.annotation.JsonbDateFormat;
import jakarta.json.bind.annotation.JsonbTransient;
import java.time.Instant;

@Entity
public class WorkItem {

    @Id
    private String id;

    @Column
    @JsonbProperty("name") // Custom field name for JSON output
    private String itemName;

    @Column
    private boolean locked;

    @Column
    private boolean done;

    @Column
    @JsonbDateFormat("yyyy-MM-dd'T'HH:mm:ss") // Customize date format for JSON serialization
    private Instant createdAt;

    @Column
    @JsonbTransient // Exclude from JSON serialization
    private Instant processedAt;

    // Default constructor required for NoSQL entity mapping
    public WorkItem() {}

    // Constructor for creating a new WorkItem
    public WorkItem(String itemName) {
        this.itemName = itemName;
        this.createdAt = Instant.now();
        this.locked = false;
        this.done = false;
    }

    // Getters and Setters
    public String getId() { return id; }
    public String getItemName() { return itemName; }
    public void setItemName(String itemName) { this.itemName = itemName; }

    public boolean isLocked() { return locked; }
    public void setLocked(boolean locked) { this.locked = locked; }

    public boolean isDone() { return done; }
    public void setDone(boolean done) { this.done = done; }

    public Instant getCreatedAt() { return createdAt; }

    public Instant getProcessedAt() { return processedAt; }
    public void setProcessedAt(Instant processedAt) { this.processedAt = processedAt; }

    @Override
    public String toString() {
        return "WorkItem{id='" + id + "', itemName='" + itemName + "', locked=" + locked +
               ", done=" + done + ", createdAt=" + createdAt + ", processedAt=" + processedAt + "}";
    }
}




import jakarta.nosql.mapping.Database;
import jakarta.nosql.mapping.Repository;
import jakarta.enterprise.context.ApplicationScoped;
import java.util.Optional;

@ApplicationScoped
public class WorkItemRepository implements Repository<WorkItem, String> {

    @Database(DatabaseType.DOCUMENT) // Use MongoDB, CouchDB, DynamoDB, etc.
    private Repository<WorkItem, String> repository;

    public WorkItem createWorkItem(String itemName) {
        WorkItem item = new WorkItem(itemName);
        return repository.save(item); // Database generates ID
    }

    public Optional<WorkItem> findById(String id) {
        return repository.findById(id);
    }
}

WorkItemRepository repo = new WorkItemRepository();

// Create a new work item with a user-defined name
WorkItem newItem = repo.createWorkItem("User-Task-001");

// Check generated ID
System.out.println("Database ID: " + newItem.getId());
System.out.println("User-defined Name: " + newItem.getItemName());

import jakarta.enterprise.concurrent.ManagedExecutorService;
import jakarta.annotation.Resource;
import java.util.concurrent.CompletableFuture;

public class WorkItemProcessor {

    @Resource
    private ManagedExecutorService executor;

    public void processAsync(WorkItem item) {
        CompletableFuture.runAsync(() -> {
            item.setLocked(true);
            item.setProcessedAt(Instant.now());
            // Save to DB
        }, executor);
    }
}
